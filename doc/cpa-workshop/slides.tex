% idea 1: the execution of loops as terms
\ideas{1}

\begin{frame}{Loops as terms}
  \[
    \begin{aligned}
      & \cur(\TRACE{x^0}{\bl{\boldsymbol{Y}}}{x, y}{u}) \\
      & \qquad \triangleq\ \FOLD{x^0}{\bl{\boldsymbol{Y}}}{x, y}{u}
    \end{aligned}
  \]
  \vspace{0.4em}
  \begin{itemize}
    \item the \concept{trace} $\tr{x^0, \dots, x^n}$ \good{retains all intermediate values}
    \item example: $\concept{\mathsf{counter}(i, c, n)} = \tr{i, i + c, \dots, i + n \cdot c}$
  \end{itemize}
  \vspace{1.5em}
  \begin{itemize}
    \item further combinators: $\mathsf{maps}$ (pointwise), $\mathsf{invs}$ (on all values)
  \end{itemize}
\end{frame}

\begin{frame}{Example: array copy}
  \begin{tikzpicture}
    \useasboundingbox (0,0) rectangle (14.4,7.4);
    \program{-0.35}{6.3}{5}{for i = 0..len, \ \ a[i] := b[i], {}, assume 0 <= j < len, assert a[j] == b[j]}
    \node[draw, very thick, anchor=north east, inner sep=5pt, font=\small] at (14.7,6.35) {%
      $\begin{aligned}
        \uncover<1->{K &= \concept{\mathsf{counter}(0, 1, n)} \qquad \text{with } 0 \le n} \\[8pt]
        \uncover<2->{I &= \TRACE{0}{\bl{K}}{i, k}{i + 1}} \\[8pt]
        \uncover<3->{A &= \mathsf{trace}\big(a^0,\ \mathsf{for}\ \underbrace{\bl{K, I}}_{\makebox[0pt]{\scriptsize dependencies}}\ \mathsf{do}\
                         \underbrace{a, k, i \mapsto a[i := b[i]]}_{\makebox[0pt]{\scriptsize update}}\big)} \\[8pt]
        \uncover<4->{& \INVS{\bl{K, I}}{k, i}{i < \mathit{len}}}
      \end{aligned}$};
  \end{tikzpicture}
\end{frame}

% ---------------------------------------------------------------------------------------------------------------------
% idea 2: named functions for computation fragments
\ideas{2}

\begin{frame}{Named functions for computation fragments}
  \begin{align*}
    \uncover<1->{& \TRACE{i}{\bl{\mathsf{counter}(0, 1, n)}}{x, k}{x + c} && \rw \concept{\mathsf{counter}(i, c, n)}} \\
    \uncover<2->{& \TRACE{a}{\bl{K}, \bl{V}}{a, k, v}{a[k := v]} && \rw \concept{\mathsf{stores}(a, \bl{K}, \bl{V})}} \\
    \uncover<3->{& \MAPS{\bl{K}}{k}{a[k]} && \rw \concept{\mathsf{selects}(a, \bl{K})}}
  \end{align*}
  \vspace{0.4em}
  \begin{itemize}
    \item<4-> defined \good{independently of the programs} in which they appear
  \end{itemize}
\end{frame}

                       \only<3>{\textcolor{cgreen}{0 \le j < n}}\only<4->{0 \le j < n}\ \mathsf{then}\ %
                       \only<1-3>{\mathsf{earlier}(\mathsf{indexof}(j, \bl{I}), \mathsf{selects}(b, \bl{I}))}%
                       \only<4>{\mathsf{earlier}(\concept{\mathsf{indexof}(j, \bl{\textcolor{cpurple}{I}})}, \mathsf{selects}(b, \bl{I}))}%

                       \only<9>{b[\textcolor{cgreen}{j}]}%

\begin{frame}{Variable-modular, not state-modular}
  \[
    x_n = \cur(X) \quad \text{where}
  \]
  \[
    X = \mathsf{trace}(x_0,\ \mathsf{for}\ \underbrace{\bl{K}, \bl{Y}}_{\text{dependencies}}\ \mathsf{do}\ x, k, y
      \mapsto \underbrace{u(x, k, y)}_{\text{update}})
  \]
  \vspace{0.3em}
  \begin{itemize}
    \item \concept{CHCs}: a loop is a relation on \emph{states}
    \item \concept{TEQs}: a loop is what each \emph{variable computes} over the iterations
    \item the loop is its own specification: \good{sound, relatively complete}
    \item no induction per program: it is in the \concept{lemmas of the database}, proved once
    \item program fragments get \good{names}: counter, stores, sums --- knowledge about them is reusable
  \end{itemize}
\end{frame}
