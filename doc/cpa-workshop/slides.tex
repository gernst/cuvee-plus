% idea 1: the execution of loops as terms
\ideas{1}

\begin{frame}{Loops as terms}
  \[
    \begin{aligned}
      & \cur(\TRACE{x^0}{\bl{\boldsymbol{Y}}}{x, y}{u}) \\
      & \qquad \triangleq\ \FOLD{x^0}{\bl{\boldsymbol{Y}}}{x, y}{u}
    \end{aligned}
  \]
  \vspace{0.4em}
  \begin{itemize}
    \item the \concept{trace} $\tr{x^0, \dots, x^n}$ \good{retains all intermediate values}
    \item example: $\concept{\mathsf{counter}(i, c, n)} = \tr{i, i + c, \dots, i + n \cdot c}$
  \end{itemize}
  \vspace{1.5em}
  \begin{itemize}
    \item further combinators: $\mathsf{maps}$ (pointwise), $\mathsf{invs}$ (on all values)
  \end{itemize}
\end{frame}

\begin{frame}{Example: array copy}
  \begin{tikzpicture}
    \useasboundingbox (0,0) rectangle (14.4,7.4);
    \program{-0.35}{6.3}{5}{for i = 0..len, \ \ a[i] := b[i], {}, assume 0 <= j < len, assert a[j] == b[j]}
    \node[draw, very thick, anchor=north east, inner sep=5pt, font=\small] at (14.7,6.35) {%
      $\begin{aligned}
        \uncover<1->{K &= \concept{\mathsf{counter}(0, 1, n)} \qquad \text{with } 0 \le n} \\[8pt]
        \uncover<2->{I &= \TRACE{0}{\bl{K}}{i, k}{i + 1}} \\[8pt]
        \uncover<3->{A &= \mathsf{trace}\big(a^0,\ \mathsf{for}\ \underbrace{\bl{K, I}}_{\makebox[0pt]{\scriptsize dependencies}}\ \mathsf{do}\
                         \underbrace{a, k, i \mapsto a[i := b[i]]}_{\makebox[0pt]{\scriptsize update}}\big)} \\[8pt]
        \uncover<4->{& \INVS{\bl{K, I}}{k, i}{i < \mathit{len}}}
      \end{aligned}$};
  \end{tikzpicture}
\end{frame}

% ---------------------------------------------------------------------------------------------------------------------
% idea 2: named functions for computation fragments
\ideas{2}

\begin{frame}{Named functions for computation fragments}
  \begin{align*}
    \uncover<1->{& \TRACE{i}{\bl{\mathsf{counter}(0, 1, n)}}{x, k}{x + c} && \rw \concept{\mathsf{counter}(i, c, n)}} \\
    \uncover<2->{& \TRACE{a}{\bl{K}, \bl{V}}{a, k, v}{a[k := v]} && \rw \concept{\mathsf{stores}(a, \bl{K}, \bl{V})}} \\
    \uncover<3->{& \MAPS{\bl{K}}{k}{a[k]} && \rw \concept{\mathsf{selects}(a, \bl{K})}}
  \end{align*}
  \vspace{0.4em}
  \begin{itemize}
    \item<4-> defined \good{independently of the programs} in which they appear
  \end{itemize}
\end{frame}

% the introduction rule of bisection (database.tex): binary search as a whole, higher-order patterns M, G, H
\begin{frame}{How to summarize binary search}
  \begin{align*}
    & \mathsf{trace}\big((l_0, h_0),\ \mathsf{for}\ \bl{K}\ \mathsf{do}\ lo, hi, k \mapsto \\
    & \qquad \mathsf{if}\ G(M(lo, hi)) \land lo < hi\ \mathsf{then}\ (M(lo, hi) + 1, hi), \\
    & \qquad \mathsf{if}\ H(M(lo, hi)) \land \lnot G(M(lo, hi)) \land lo < hi\ \mathsf{then}\ (lo, M(lo, hi)), \\
    & \qquad \mathsf{if}\ \textit{(otherwise)}\ \mathsf{then}\ (lo, hi)\big) \\
    & \quad \rw \concept{\mathsf{bisection}((l_0, h_0), \bl{K}, M, G, H)}
  \end{align*}
  \vspace{0.6em}
  \begin{itemize}
    \item expressing lemmas like this points to required features (forwarding, higher-order, \dots)
    \item lesson: engine and lemma database were \concept{co-developed}
  \end{itemize}
\end{frame}

% related work (related.tex): a few principles, each against TEQs
\begin{frame}{Related work}
  \small
  \begin{itemize}
    \setlength{\itemsep}{0.4em}
    \item \concept{loops as axioms} over time points \cit{[Trace Logic, VIAP]}\\
          TEQs: a trace is a \good{term}, lemmas are matched against it
    \item \concept{loop acceleration}, also for arrays \cit{[Frohn, Giesl; Booster]}\\
          TEQs: acceleration schemes are \good{lemmas}, not built into the algorithm
    \item \concept{quantified invariants} \cit{[Horn clauses, synthesis]}\\
          TEQs: \good{no quantifier to guess}, the trace is the whole history
    \item \concept{folds} in decidable logics \cit{[Array Folds Logic]}\\
          TEQs: give up decidability, gain \good{compositionality}
    \item \concept{LLM invariants} per task \cit{[Lemur, LaM4Inv]}\\
          TEQs: lemmas are \good{written once, reused, and proved}
  \end{itemize}
\end{frame}
