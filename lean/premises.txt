# the premises of the lemmas of the database that the engine does not check, which the generator adds to the printed
# statements (scripts/lean/generate.py): a line `name: premise -- why` adds `premise →` before the statement, a premise
# in parentheses, `name: (h : φ) -- why`, a further binder; the lemmas are sound only in the contexts that provide the
# premises, which those of -teq do as stated in the reasons; each line is a gap between the engine and the semantics

# reads within the loop: -teq reads the lists of a loop of n iterations at its iterations k < n only, earlier(natural(k),
# xs) for a counter k from 0 by 1 in the iteration constraints, and the elements outside of a list are unspecified
earlier_counter: integer k < integer n -- the reads of a counter are at the iterations of its loop
earlier_consts: integer k < integer n -- the reads of a constant trace are at the iterations of its loop
earlier_selects: integer j < integer n -- the reads of the selects of a loop are at its iterations
earlier_stores: integer j < integer n -- the earlier states of a loop are read at its iterations
earlier_sums: integer j < integer n -- the earlier states of a loop are read at its iterations
earlier_counts: integer j < integer n -- the earlier states of a loop are read at its iterations
current_maxs_earlier: integer k < integer n -- the values of a loop are read at its iterations
current_mins_earlier: integer k < integer n -- the values of a loop are read at its iterations

# prefixes: the prefixes that the lemmas make are those of the earlier states of a loop, at most its length
prefix_counter: integer k ≤ integer n -- the prefixes of a counter are cut at the iterations of its loop

# lengths that lean.Printer.lengths does not see, since the list is no dependency of a combinator
sum_consts_prefix: (hjs : length js = n) -- the values of an inner loop, one per iteration of the outer one

# search.dfy
earlier_not_occurs_prefix: (hxs : length xs = n) -- xs is a list of the loop of n iterations

# counters.dfy: arbitrary guards of the cases of an update, whose completeness the module does not state; the
# printer ends their conditionals in unspecified (lean.Printer)
current_trace_bounded: (hc : ∀ (y : Int), ¬C1 y ∧ ¬C2 y → C3 y) -- the guards of the cases of an update are complete (doc/teq.md), the loop has no break
current_trace_increasing: (hc : ∀ (h : Int), C h ∨ D h ∨ E h) -- the guards of the cases of an update are complete (doc/teq.md), the loop has no break
