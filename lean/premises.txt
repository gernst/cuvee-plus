# the premises of the lemmas of the database that the engine does not check, which the generator adds to the printed
# statements (scripts/lean/generate.py): a line `name: premise -- why` adds `premise →` before the statement, a premise
# in parentheses, `name: (h : φ) -- why`, a further binder; the lemmas are sound only in the contexts that provide the
# premises, which those of -teq do as stated in the reasons; each line is a gap between the engine and the semantics

# reads within the loop: -teq reads the lists of a loop of n iterations at its iterations 0 <= k < n only, earlier(k,
# xs) for a counter k from 0 by 1 in the iteration constraints, and the elements outside of a list (also at negative
# positions) are unspecified
earlier_counter: 0 ≤ k ∧ k < n -- the reads of a counter are at the iterations of its loop
earlier_consts: 0 ≤ k ∧ k < n -- the reads of a constant trace are at the iterations of its loop
earlier_selects: 0 ≤ j ∧ j < n -- the reads of the selects of a loop are at its iterations
earlier_stores: 0 ≤ j ∧ j < n -- the earlier states of a loop are read at its iterations
earlier_sums: 0 ≤ j ∧ j < n -- the earlier states of a loop are read at its iterations
earlier_counts: 0 ≤ j ∧ j < n -- the earlier states of a loop are read at its iterations
current_maxs_earlier: 0 ≤ k ∧ k < n -- the values of a loop are read at its iterations
current_mins_earlier: 0 ≤ k ∧ k < n -- the values of a loop are read at its iterations
current_maxs_prefix_earlier: (hxs : k < length xs) -- the list is read at the iterations of a loop that it covers
current_mins_prefix_earlier: (hxs : k < length xs) -- the list is read at the iterations of a loop that it covers

# prefixes: the prefixes that the lemmas make are those of the earlier states of a loop, at most its length
prefix_counter: k ≤ n -- the prefixes of a counter are cut at the iterations of its loop

# lengths that lean.Printer.lengths does not see, since the list is no dependency of a combinator
earlier_not_occurs_prefix: (hxs : length xs = n) -- xs is a list of the loop of n iterations
earlier_above_iterations: (hn : n1 = n) -- ts is a list of the loop of n iterations, the dependency of the fold above_iterations
